## Usage

```tsx title="components/app-sidebar.tsx"
import {
  Sidebar,
  SidebarContent,
  SidebarFooter,
  SidebarGroup,
  SidebarHeader,
} from "@/components/ui/sidebar"

export const AppSidebar = () => {
  return (
    <Sidebar>
      <SidebarHeader />
      <SidebarContent>
        <SidebarGroup />
        <SidebarGroup />
      </SidebarContent>
      <SidebarFooter />
    </Sidebar>
  )
}
```

## Controlled

Use `open` / `onOpenChange` to control the sidebar.

```tsx
import React from "react";
import { Sidebar, SidebarProvider } from "@/components/ui/sidebar";

export const ControlledLayout = () => {
  const [open, setOpen] = React.useState(false);

  return (
    <SidebarProvider onOpenChange={({ open }) => setOpen(open)} open={open}>
      <Sidebar />
    </SidebarProvider>
  );
};
```

## Collapsible

Use `collapsible` to control the collapsibility of the sidebar.

### Icon

Collapse the sidebar to icons.

<PreviewIframe src="/view/examples/sidebar/example-collapsible-icon" title="Icon" />

```tsx

  <Sidebar collapsible="icon">
    <SidebarContent>
      <SidebarGroup>
        <SidebarGroupLabel>Workspace</SidebarGroupLabel>
        <SidebarMenu>
          <SidebarMenuItem>
```

```tsx
const SidebarState = () => {
  const { state, toggleSidebar } = useSidebar();
```

## API Reference

### SidebarProvider

Provides open/collapsed state.

| Prop | Type | Default |
| --- | --- | --- |
| `defaultOpen` | `boolean` | `true` |
| `defaultOpenMobile` | `boolean` | `false` |
| `open` | `boolean` | `-` |
| `onOpenChange` | `({ open }: { open: boolean }) => void` | `-` |

| Prop | Type | Default |
| --- | --- | --- |
| `--sidebar-width` | `string` | `16rem` |
| `--sidebar-width-icon` | `string` | `3rem` |

### Sidebar

The sidebar column. Fixed on desktop, sheet on mobile.

- `offcanvas` slides off screen
- `icon` collapses to icon width
- `none` stays expanded

| Prop | Type | Default |
| --- | --- | --- |
| `placement` | `"left" \| "right"` | `"left"` |
| `variant` | `"sidebar" \| "floating" \| "inset"` | `"sidebar"` |
| `collapsible` | `"offcanvas" \| "icon" \| "none"` | `"offcanvas"` |
| `className` | `string` | `-` |

### SidebarInset

Main content beside the sidebar. Required for `variant="inset"`.

| Prop | Type | Default |
| --- | --- | --- |
| `asChild` | `boolean` | `false` |
| `className` | `string` | `-` |

### useSidebar

Access sidebar state and actions.

| Prop | Type | Default |
| --- | --- | --- |
| `state` | `"expanded" \| "collapsed"` | `-` |
| `open` | `boolean` | `-` |
| `setOpen` | `(value: boolean \| ((value: boolean) => boolean)) => void` | `-` |
| `openMobile` | `boolean` | `-` |
| `setOpenMobile` | `(open: boolean) => void` | `-` |
| `isMobile` | `boolean` | `-` |
| `toggleSidebar` | `() => void` | `-` |

### SidebarHeader

Top section. Brand, workspace switcher, etc.

| Prop | Type | Default |
| --- | --- | --- |
| `asChild` | `boolean` | `false` |
| `className` | `string` | `-` |

### SidebarFooter

| Prop | Type | Default |

### SidebarSeparator

| Prop | Type | Default |
