```text
│   └── StateDescription
└── StateContent
```

## Usage

The following composition also uses [Button](/docs/components/button) and Lucide icons. Install `@shark/button` and `lucide-react` before copying it.

```tsx
import { FolderCodeIcon } from "lucide-react";
import { Button } from "@/components/ui/button";
import {
  State,
  StateContent,
  StateDescription,
  StateHeader,
  StateMedia,
  StateTitle,
} from "@/components/ui/state";
```

```tsx
<State>
  <StateHeader>
    <StateMedia variant="icon">
      <FolderCodeIcon aria-hidden="true" />
    </StateMedia>
    <StateTitle asChild>
      <h2>No projects yet</h2>
    </StateTitle>
    <StateDescription>Create a project to get started.</StateDescription>
  </StateHeader>
  <StateContent>
    <Button>Create project</Button>
  </StateContent>
</State>
```

## API Reference

### State

The root layout for a contextual screen state. Wraps `StateHeader` and `StateContent`.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | - |
| `asChild` | `boolean` | `false` |

### StateHeader

Wraps the state media, title, and description.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | - |
| `asChild` | `boolean` | `false` |

### StateMedia

Displays media such as an icon, image, avatar, or custom element.

| Prop | Type | Default |
| --- | --- | --- |
| `variant` | `"default" \| "icon"` | `"default"` |
| `className` | `string` | - |
| `asChild` | `boolean` | `false` |

### StateTitle

Displays the state title. Use `asChild` to supply the heading level appropriate to the page hierarchy.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | - |
| `asChild` | `boolean` | `false` |

### StateDescription

Displays supporting state text.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | - |
| `asChild` | `boolean` | `false` |

### StateContent

Holds next actions or supporting content such as buttons, inputs, and links.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | - |
| `asChild` | `boolean` | `false` |
