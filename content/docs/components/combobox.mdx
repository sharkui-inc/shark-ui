## Usage

```tsx
import { useFilter, useListCollection } from "@ark-ui/react";
import {
  Combobox,
  ComboboxContent,
  ComboboxEmpty,
  ComboboxGroup,
  ComboboxInput,
  ComboboxItem,
  ComboboxList,
} from "@/components/ui/combobox";
```

```tsx
const { contains } = useFilter({ sensitivity: "base" });
const { collection, filter } = useListCollection({
  initialItems: [
    { label: "Apple", value: "apple" },
    { label: "Banana", value: "banana" },
  ],
  filter: contains,
});

<Combobox
  collection={collection}
  onInputValueChange={({ inputValue, reason }) =>
    filter(reason === "item-select" ? "" : inputValue)
  }
>
  <ComboboxInput placeholder="Select an option" />
  <ComboboxContent>
    <ComboboxEmpty />
    <ComboboxList>
      {items.map((item) => (
        <ComboboxItem item={item} key={item.value}>
          {item.label}
        </ComboboxItem>
      ))}
    </ComboboxList>
  </ComboboxContent>
</Combobox>
```

## API Reference

### Combobox

Root component.

| Prop | Type | Default |
| --- | --- | --- |
| `collection` | `ListCollection<T>` | **required** |
| `value` | `string \| string[]` | - |
| `defaultValue` | `string \| string[]` | - |
| `onValueChange` | `(details: ValueChangeDetails<T>) => void` | - |
| `onInputValueChange` | `(details: InputValueChangeDetails) => void` | - |
| `multiple` | `boolean` | `false` |
| `selectionBehavior` | `"clear" \| "replace" \| "preserve"` | `"replace"` |
| `disabled` | `boolean` | - |
| `openOnClick` | `boolean` | `true` |
| `alwaysSubmitOnEnter` | `boolean` | - |
| `lazyMount` | `boolean` | `true` |
| `unmountOnExit` | `boolean` | `true` |
| `className` | `string` | - |

### ComboboxControl

Optional wrapper for custom layouts.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | - |
| `asChild` | `boolean` | - |

### ComboboxInput

Input with menu trigger and optional clear button. Combines search and selection.

| Prop | Type | Default |
| --- | --- | --- |
| `size` | `"sm" \| "md" \| "lg"` | `"md"` |
| `variant` | `"ghost"` | - |
| `showTrigger` | `boolean` | `true` |
| `showClear` | `boolean` | `false` |
| `disabled` | `boolean` | - |
| `className` | `string` | - |

### ComboboxChips

Control for multiple selections. Wrap `ComboboxChip` values and a
`ComboboxChipsInput` inside it.

| Prop | Type | Default |
| --- | --- | --- |
| `size` | `"sm" \| "md" \| "lg"` | `"md"` |
| `className` | `string` | - |

### ComboboxChipsInput

| Prop | Type | Default |
| --- | --- | --- |
| `placeholder` | `string` | - |
| `className` | `string` | - |

Displays one selected value and removes it from the combobox when its button is pressed.

| Prop | Type | Default |
| --- | --- | --- |
| `value` | `string` | **required** |
| `children` | `ReactNode` | **required** |
| `removeProps` | `ComponentProps<"button">` | - |
| `className` | `string` | - |

### ComboboxTrigger

| Prop | Type | Default |
| --- | --- | --- |
