## Usage

```tsx
import { useState } from "react";
import { PowerSearch, usePowerSearchConfig } from "@/components/ui/power-search";
import type { PowerSearchFilter } from "@/components/ui/power-search";
```

```tsx
const config = usePowerSearchConfig([
  { key: "title", type: "string", label: "Title" },
  { key: "year", type: "number", label: "Publication Year" },
]);

const [filters, setFilters] = useState<readonly PowerSearchFilter[]>([]);
const filteredData = config.applyFilters(mockData, filters);

<PowerSearch
  config={config}
  filters={filters}
  onChange={setFilters}
  placeholder="Search..."
/>
```

## API Reference

### Power Search

The main UI component.

| Prop | Type | Default |
| --- | --- | --- |
| `config` | `PowerSearchConfig` | **required** |
| `filters` | `readonly PowerSearchFilter[]` | **required** |
| `onChange` | `(filters: readonly PowerSearchFilter[]) => void` | **required** |
| `placeholder` | `string` | `"Search..."` |
| `resultCount` | `number \| string` | - |
| `disabled` | `boolean` | `false` |
| `readOnly` | `boolean` | `false` |
| `className` | `string` | - |

### usePowerSearchConfig

A hook to build the configuration object.

```tsx
function usePowerSearchConfig(
  definitions: ReadonlyArray<FieldDefinition>,
  name?: string
): PowerSearchConfig
```
