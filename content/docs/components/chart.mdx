## Usage

```tsx
import { Bar, BarChart } from "recharts"
import { ChartContainer, ChartTooltip, ChartTooltipContent } from "@/components/ui/chart"

export const MyChart = () => {
  return (
    <ChartContainer>
      <BarChart data={data}>
        <Bar dataKey="value" />
        <ChartTooltip content={<ChartTooltipContent />} />
      </BarChart>
    </ChartContainer>
  )
}
```

### Start with data

Each row is a month. `dataKey` maps fields to series:

```tsx
const chartData = [
  { month: "January", desktop: 186, mobile: 80 },
  { month: "February", desktop: 305, mobile: 200 },
  { month: "March", desktop: 237, mobile: 120 },
  { month: "April", desktop: 73, mobile: 190 },
  { month: "May", desktop: 209, mobile: 130 },
  { month: "June", desktop: 214, mobile: 140 },
]
```

### Add a chart config

`ChartConfig` holds labels, optional icons, and colors. Keep it separate from the data so charts can share it.

```tsx
import { type ChartConfig } from "@/components/ui/chart"

const chartConfig = {
  desktop: {
    label: "Desktop",
    color: "#2563eb",
  },
  mobile: {
    label: "Mobile",
    color: "#60a5fa",
  },
} satisfies ChartConfig
```

### Render the chart

Pass `config` to `ChartContainer`. Set a height (`min-h-[200px]` or a fixed `h-*`) or keep the default `aspect-video` so Recharts can size the SVG.

```tsx
import { Bar, BarChart } from "recharts"
import { ChartContainer, type ChartConfig } from "@/components/ui/chart"

<ChartContainer config={chartConfig} className="min-h-[200px] w-full">
  <BarChart accessibilityLayer data={chartData}>
    <Bar dataKey="desktop" fill="var(--color-desktop)" radius={4} />
    <Bar dataKey="mobile" fill="var(--color-mobile)" radius={4} />
  </BarChart>
</ChartContainer>
```

### Add a grid

Add `CartesianGrid` from Recharts. `vertical={false}` draws horizontal lines only:

```tsx
import { Bar, BarChart, CartesianGrid } from "recharts"

<ChartContainer config={chartConfig} className="min-h-[200px] w-full">
  <BarChart accessibilityLayer data={chartData}>
    <CartesianGrid vertical={false} />
    <Bar dataKey="desktop" fill="var(--color-desktop)" radius={4} />
    <Bar dataKey="mobile" fill="var(--color-mobile)" radius={4} />
  </BarChart>
</ChartContainer>
```

### Add an axis

Use `XAxis` for categories. This example shortens month labels and hides tick and axis lines:

```tsx
import { Bar, BarChart, CartesianGrid, XAxis } from "recharts"

<ChartContainer config={chartConfig} className="h-[200px] w-full">
  <BarChart accessibilityLayer data={chartData}>
    <CartesianGrid vertical={false} />
    <XAxis
      dataKey="month"
      tickLine={false}
      tickMargin={10}
      axisLine={false}
      tickFormatter={(value) => value.slice(0, 3)}
    />
    <Bar dataKey="desktop" fill="var(--color-desktop)" radius={4} />
    <Bar dataKey="mobile" fill="var(--color-mobile)" radius={4} />
  </BarChart>
</ChartContainer>
```

### Add a tooltip

`ChartTooltip` and `ChartTooltipContent` take labels, colors, and values from `chartConfig`:

```tsx
import { ChartTooltip, ChartTooltipContent } from "@/components/ui/chart"

<ChartContainer config={chartConfig} className="h-[200px] w-full">
  <BarChart accessibilityLayer data={chartData}>
    <CartesianGrid vertical={false} />
    <XAxis
      dataKey="month"
      tickLine={false}
      tickMargin={10}
      axisLine={false}
      tickFormatter={(value) => value.slice(0, 3)}
    />
    <ChartTooltip content={<ChartTooltipContent />} />
    <Bar dataKey="desktop" fill="var(--color-desktop)" radius={4} />
    <Bar dataKey="mobile" fill="var(--color-mobile)" radius={4} />
  </BarChart>
</ChartContainer>
```

### Add a legend

`ChartLegend` and `ChartLegendContent` use the same config:

```tsx
import { ChartLegend, ChartLegendContent } from "@/components/ui/chart"

<ChartContainer config={chartConfig} className="h-[200px] w-full">
  <BarChart accessibilityLayer data={chartData}>
    <CartesianGrid vertical={false} />
    <XAxis
      dataKey="month"
      tickLine={false}
      tickMargin={10}
      axisLine={false}
      tickFormatter={(value) => value.slice(0, 3)}
    />
    <ChartTooltip content={<ChartTooltipContent />} />
    <ChartLegend content={<ChartLegendContent />} />
    <Bar dataKey="desktop" fill="var(--color-desktop)" radius={4} />
    <Bar dataKey="mobile" fill="var(--color-mobile)" radius={4} />
  </BarChart>
</ChartContainer>
```

## Chart config

`ChartConfig` maps each series key to a label, optional icon, and either `color` or a `theme` object with `light` and `dark` values.

```tsx
import { Monitor } from "lucide-react"
import { type ChartConfig } from "@/components/ui/chart"

const chartConfig = {
  desktop: {
    label: "Desktop",
    icon: Monitor,
    color: "#2563eb",
    theme: {
      light: "#2563eb",
      dark: "#dc2626",
    },
  },
} satisfies ChartConfig
```

Use `color` when both themes share a value. Use `theme` when they differ.

## Theming

Chart uses CSS variables to style series. See the full palette in [Colors](/docs/colors).

### Variable Reference

- <div className="inline-block size-5 rounded border border-border bg-chart-1 me-2 align-middle" /> `--chart-1`: first series
- <div className="inline-block size-5 rounded border border-border bg-chart-2 me-2 align-middle" /> `--chart-2`: second series
- <div className="inline-block size-5 rounded border border-border bg-chart-3 me-2 align-middle" /> `--chart-3`: third series
- <div className="inline-block size-5 rounded border border-border bg-chart-4 me-2 align-middle" /> `--chart-4`: fourth series
- <div className="inline-block size-5 rounded border border-border bg-chart-5 me-2 align-middle" /> `--chart-5`: fifth series

Learn more about styling in the [Styling](/docs/styling) section.

Set `color` to those variables so series follow the theme:

```tsx
const chartConfig = {
  desktop: {
    label: "Desktop",
    color: "var(--chart-1)",
  },
  mobile: {
    label: "Mobile",
    color: "var(--chart-2)",
  },
} satisfies ChartConfig
```

### Hex, HSL, or OKLCH

You can also pass hex, HSL, OKLCH, or mix them with variables:

```tsx
const chartConfig = {
  desktop: { label: "Desktop", color: "#2563eb" },
  mobile: { label: "Mobile", color: "hsl(220, 98%, 61%)" },
  tablet: { label: "Tablet", color: "oklch(0.5 0.2 240)" },
  laptop: { label: "Laptop", color: "var(--chart-2)" },
} satisfies ChartConfig
```

### Applying colors in markup and data

`ChartContainer` sets `var(--color-<key>)` for each config key. Use it on fills, in the data, or in Tailwind:

```tsx
<Bar dataKey="desktop" fill="var(--color-desktop)" />

{ browser: "chrome", visitors: 275, fill: "var(--color-chrome)" }

<LabelList className="fill-(--color-desktop)" />
```

## Tooltip

```tsx
import { ChartTooltip, ChartTooltipContent } from "@/components/ui/chart"

<ChartTooltip content={<ChartTooltipContent />} />
```

### Custom label and name keys

Pass `labelKey` and `nameKey` when the tooltip header and row titles should come from other fields:

```tsx
const chartData = [
  { browser: "chrome", visitors: 187, fill: "var(--color-chrome)" },
  { browser: "safari", visitors: 200, fill: "var(--color-safari)" },
]

const chartConfig = {
  visitors: {
    label: "Total Visitors",
  },
  chrome: {
    label: "Chrome",
    color: "var(--chart-1)",
  },
  safari: {
    label: "Safari",
    color: "var(--chart-2)",
  },
} satisfies ChartConfig

<ChartTooltip
  content={<ChartTooltipContent labelKey="visitors" nameKey="browser" />}
/>
```

## Legend

`ChartLegendContent` builds legend items from `chartConfig`.

```tsx
import { ChartLegend, ChartLegendContent } from "@/components/ui/chart"

<ChartLegend content={<ChartLegendContent />} />
```

### Custom name key

Pass `nameKey` when legend labels should come from a field on the data (such as `browser`):

```tsx
const chartData = [
  { browser: "chrome", visitors: 187, fill: "var(--color-chrome)" },
  { browser: "safari", visitors: 200, fill: "var(--color-safari)" },
]

const chartConfig = {
  chrome: {
    label: "Chrome",
    color: "var(--chart-1)",
  },
  safari: {
    label: "Safari",
    color: "var(--chart-2)",
  },
} satisfies ChartConfig

<ChartLegend content={<ChartLegendContent nameKey="browser" />} />
```

## Accessibility

Pass `accessibilityLayer` for keyboard and screen reader support. Skip it on decorative charts that should not take focus.

```tsx
<BarChart accessibilityLayer data={chartData}>
  {/* children */}
</BarChart>
```

```tsx
<LineChart accessibilityLayer data={chartData}>
  {/* children */}
</LineChart>
```
