## Usage

```tsx
import { createColumnHelper } from "@tanstack/react-table";
import {
  DataTable,
  DataTableColumnHeader,
  type DataTableFeatures,
} from "@/components/ui/data-table";
import { FormatNumber } from "@/components/ui/format";

type Payment = {
  amount: number;
  email: string;
  id: string;
  status: "pending" | "processing" | "success" | "failed";
};

const columnHelper = createColumnHelper<DataTableFeatures, Payment>();

const columns = columnHelper.columns([
  columnHelper.accessor("status", {
    header: "Status",
  }),
  columnHelper.accessor("email", {
    header: "Email",
  }),
  columnHelper.accessor("amount", {
    cell: ({ getValue }) => (
      <FormatNumber currency="USD" style="currency" value={getValue()} />
    ),
    header: ({ column }) => (
      <DataTableColumnHeader column={column} title="Amount" />
    ),
  }),
]);

export function PaymentsTable({ data }: { data: Payment[] }) {
  return (
    <DataTable
      caption="Recent payments"
      columns={columns}
      data={data}
    />
  );
}
```

## Features

`dataTableFeatures` is exported so custom compositions can use exactly the same v9 feature set with `useTable`. TanStack Table includes the core row model by default; the optional row models are registered by this component with `createFilteredRowModel`, `createSortedRowModel`, and `createPaginatedRowModel`.

```tsx
import { useTable } from "@tanstack/react-table";
import { dataTableFeatures } from "@/components/ui/data-table";

const table = useTable({
  columns,
  data,
  features: dataTableFeatures,
});
```

Use `DataTableColumnHeader` in a column header for an Ark UI menu with ascending, descending, and hide actions.

## Pagination and view options

For a custom table composition created with `useTable`, pass its instance to `DataTablePagination` and `DataTableViewOptions`.

```tsx
import {
  DataTablePagination,
  DataTableViewOptions,
} from "@/components/ui/data-table";

<div className="flex items-center gap-2">
  <DataTableViewOptions table={table} />
</div>

<DataTablePagination table={table} />
```

## API Reference

### DataTable

| Prop | Type | Default |
| --- | --- | --- |
| `columns` | `ColumnDef<DataTableFeatures, TData>[]` | - |
| `data` | `TData[]` | - |
| `caption` | `ReactNode` | - |
| `emptyMessage` | `ReactNode` | `"No results."` |
| `tableOptions` | `TableOptions` without `columns`, `data`, and `features` | - |
| `className` | `string` | - |

### DataTableColumnHeader

| Prop | Type | Default |
| --- | --- | --- |
| `column` | `Column<DataTableFeatures, TData, TValue>` | - |
| `title` | `string` | - |
| `className` | `string` | - |

### DataTablePagination

| Prop | Type | Default |
| --- | --- | --- |
| `table` | `ReactTable<DataTableFeatures, TData>` | - |
| `className` | `string` | - |

### DataTableViewOptions

| Prop | Type | Default |
| --- | --- | --- |
| `table` | `ReactTable<DataTableFeatures, TData>` | - |
| `className` | `string` | - |
