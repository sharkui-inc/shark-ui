## Usage

```tsx
import { createColumnHelper } from "@tanstack/react-table";
import {
  DataTable,
  DataTableColumnHeader,
  type DataTableFeatures,
} from "@/components/ui/data-table";
import { FormatNumber } from "@/components/ui/format";

type Payment = {
  amount: number;
  email: string;
  id: string;
  status: "pending" | "processing" | "success" | "failed";
};

const columnHelper = createColumnHelper<DataTableFeatures, Payment>();

const columns = columnHelper.columns([
  columnHelper.accessor("status", {
    header: "Status",
  }),
  columnHelper.accessor("email", {
    header: "Email",
  }),
  columnHelper.accessor("amount", {
    cell: ({ getValue }) => (
      <FormatNumber currency="USD" style="currency" value={getValue()} />
    ),
    header: ({ column }) => (
      <DataTableColumnHeader column={column} title="Amount" />
    ),
  }),
]);

export function PaymentsTable({ data }: { data: Payment[] }) {
  return (
    <DataTable
      caption="Recent payments"
      columns={columns}
      data={data}
    />
  );
}
```

## API Reference

### DataTable

| Prop | Type | Default |
| --- | --- | --- |
| `columns` | `ColumnDef<DataTableFeatures, TData>[]` | - |
| `data` | `TData[]` | - |
| `caption` | `ReactNode` | - |
| `emptyMessage` | `ReactNode` | `"No results."` |
| `tableOptions` | `TableOptions` without `columns`, `data`, and `features` | - |
| `className` | `string` | - |

### DataTableColumnHeader

| Prop | Type | Default |
| --- | --- | --- |
| `column` | `Column<DataTableFeatures, TData, TValue>` | - |
| `title` | `string` | - |
| `className` | `string` | - |

### DataTablePagination

| Prop | Type | Default |
| --- | --- | --- |
| `table` | `ReactTable<DataTableFeatures, TData>` | - |
| `className` | `string` | - |

### DataTableViewOptions

| Prop | Type | Default |
| --- | --- | --- |
| `table` | `ReactTable<DataTableFeatures, TData>` | - |
| `className` | `string` | - |
