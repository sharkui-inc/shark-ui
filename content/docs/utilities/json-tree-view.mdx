## Usage

```tsx
import { JsonTreeView } from "@/registry/react/components/json-tree-view";

const orderFulfilledEvent = {
  event: "order.fulfilled",
  order: {
    id: "ord_01K5H2Q4J6FM3A",
    total: 12900,
  },
};

<JsonTreeView data={orderFulfilledEvent} defaultExpandedDepth={1} />
```
