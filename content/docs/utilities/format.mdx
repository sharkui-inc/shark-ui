## Usage

```tsx
import { 
  FormatByte, 
  FormatNumber, 
  FormatRelativeTime,
  FormatTime,
} from "@/components/ui/format";
```

```tsx
<FormatByte value={120_000} />
<FormatNumber value={1234567} />
<FormatRelativeTime value={new Date("2025-05-05")} />
<FormatTime value={new Date("2026-08-24T14:30:00")} />
```

## API Reference

### FormatByte

| Prop | Type | Default |
| --- | --- | --- |
| `value` | `number` | `-` |
| `unit` | `"byte" \| "bit"` | `"byte"` |
| `unitDisplay` | `"long" \| "short" \| "narrow"` | `-` |
| `unitSystem` | `"decimal" \| "binary"` | `-` |
| `locale` | `string` | `-` |

### FormatRelativeTime

| Prop | Type | Default |
| --- | --- | --- |
| `value` | `Date \| number` | `-` |
| `numeric` | `"always" \| "auto"` | `-` |
| `style` | `"long" \| "short" \| "narrow"` | `-` |
| `locale` | `string` | `-` |

### FormatTime

| Prop | Type | Default |
| --- | --- | --- |
| `value` | `Date \| string` | `-` |
| `format` | `"12h" \| "24h"` | `-` |
| `amLabel` | `string` | `-` |
| `pmLabel` | `string` | `-` |
| `withSeconds` | `boolean` | `-` |
| `locale` | `string` | `-` |

### FormatNumber

| Prop | Type | Default |
| --- | --- | --- |
| `value` | `number` | `-` |
| `locale` | `string` | `-` |
| `style` | `"decimal" \| "percent" \| "currency" \| "unit"` | `"decimal"` |
| `currency` | `string` | `-` |
| `unit` | `string` | `-` |
| `notation` | `"standard" \| "compact" \| "scientific" \| "engineering"` | `"standard"` |
| `minimumFractionDigits` | `number` | `-` |
| `maximumFractionDigits` | `number` | `-` |
