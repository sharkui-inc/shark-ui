## Usage

```tsx
import { createListCollection } from "@ark-ui/react/collection";
import { useListSelection } from "@/hooks/use-list-selection";
```

```tsx
const collection = createListCollection({
  items: [
    { label: "React", value: "react" },
    { label: "Vue", value: "vue" },
  ],
});

const selection = useListSelection({ collection });
```

## API Reference

### useListSelection

| Prop | Type | Default |
| --- | --- | --- |
| `collection` | `ListCollection<T>` | `required` |
| `selectionMode` | `"single" \| "multiple" \| "none"` | `"single"` |
| `deselectable` | `boolean` | `true` |
| `initialSelectedValues` | `string[]` | `[]` |
| `resetOnCollectionChange` | `boolean` | `false` |

### Return value

| Prop | Type | Default |
| --- | --- | --- |
| `selectedValues` | `string[]` | `-` |
| `isEmpty` | `boolean` | `-` |
| `firstSelectedValue` | `string \| null` | `-` |
| `lastSelectedValue` | `string \| null` | `-` |
| `isSelected` | `(value: string \| null) => boolean` | `-` |
| `canSelect` | `(value: string) => boolean` | `-` |
| `isAllSelected` | `() => boolean` | `-` |
| `isSomeSelected` | `() => boolean` | `-` |
| `select` | `(value: string, forceToggle?: boolean) => void` | `-` |
| `deselect` | `(value: string) => void` | `-` |
| `toggle` | `(value: string) => void` | `-` |
| `replace` | `(value: string \| null) => void` | `-` |
| `extend` | `(anchorValue: string, targetValue: string) => void` | `-` |
| `setSelectedValues` | `(values: string[]) => void` | `-` |
| `clear` | `() => void` | `-` |
