## Usage

```tsx
import {
  MessageBubble,
  MessageBubbleContent,
  MessageBubbleGroup,
  MessageBubbleReactions,
  MessageBubbleTrigger,
} from "@/components/ui/message-bubble";
```

```tsx
<MessageBubbleGroup>
  <MessageBubble variant="secondary">
    <MessageBubbleContent>You still coming Thursday?</MessageBubbleContent>
  </MessageBubble>
  <MessageBubble variant="secondary">
    <MessageBubbleContent>
      Door's unlocked if you get here first.
    </MessageBubbleContent>
  </MessageBubble>
</MessageBubbleGroup>
<MessageBubbleGroup>
  <MessageBubble align="end">
    <MessageBubbleContent>Yeah, around 7.</MessageBubbleContent>
  </MessageBubble>
  <MessageBubble align="end">
    <MessageBubbleContent>I'll grab ice.</MessageBubbleContent>
    <MessageBubbleReactions aria-label="Reactions: thumbs up" role="img">
      <span>👍</span>
    </MessageBubbleReactions>
  </MessageBubble>
</MessageBubbleGroup>
```

## API Reference

### MessageBubbleGroup

Stacks related bubbles.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageBubble

Message surface. 

| Prop | Type | Default |
| --- | --- | --- |
| `variant` | `"default" \| "secondary" \| "tinted" \| "outline" \| "ghost" \| "destructive"` | `"default"` |
| `align` | `"start" \| "end"` | `"start"` |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageBubbleContent

Body of the bubble.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageBubbleTrigger

Hover-revealed control in the top-end corner.

| Prop | Type | Default |
| --- | --- | --- |
| `aria-label` | `string` | `"Message actions"` |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageBubbleReactions

Reaction cluster anchored to the bubble.

| Prop | Type | Default |
| --- | --- | --- |
| `side` | `"top" \| "bottom"` | `"bottom"` |
| `align` | `"start" \| "end"` | `"end"` |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |
