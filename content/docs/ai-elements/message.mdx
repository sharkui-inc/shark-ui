```text
└── Message

        ├── MessageHeader

        └── MessageFooter
            └── MessageActions

```

## Usage

```tsx
import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar";
import {
  Message,
  MessageAvatar,
  MessageContent,
} from "@/components/ui/message";
import {
  MessageBubble,
  MessageBubbleContent,
} from "@/components/ui/message-bubble";
```

```tsx
<Message>
  <MessageAvatar>
    <Avatar>
      <AvatarImage alt="@vinihvc" src="https://github.com/vinihvc.png" />
      <AvatarFallback>VV</AvatarFallback>
    </Avatar>
  </MessageAvatar>
  <MessageContent>
    <MessageBubble>
      <MessageBubbleContent>How can I help you today?</MessageBubbleContent>
    </MessageBubble>
  </MessageContent>
</Message>
```

The row is `Message`. Put the surface in [Message Bubble](/docs/ai-elements/message-bubble), set `align="end"` for the current user, and scroll the thread with [Message Scroller](/docs/ai-elements/message-scroller).

## API Reference

### MessageGroup

Stacks related messages.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### Message

Transcript row. `align="end"` is the current user.

| Prop | Type | Default |
| --- | --- | --- |
| `align` | `"start" \| "end"` | `"start"` |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageAvatar

Bottom of the bubble. Shifts up past a footer or bottom reactions.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageContent

Column for header, bubble, and footer.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageHeader

Name or timestamp above the bubble.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageFooter

Timestamp or actions below the bubble.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageActions

Container for message action buttons.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### MessageAction

Icon button with an optional tooltip.

| Prop | Type | Default |
| --- | --- | --- |
| `tooltip` | `string` | `-` |
| `label` | `string` | `-` |
| `variant` | `"default" \| "outline" \| "secondary" \| "ghost" \| "link" \| "destructive"` | `"ghost"` |
| `size` | `"xs" \| "sm" \| "md" \| "lg" \| "xl" \| "icon-xs" \| "icon-sm" \| "icon-md" \| "icon-lg" \| "icon-xl"` | `"icon-xs"` |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |
