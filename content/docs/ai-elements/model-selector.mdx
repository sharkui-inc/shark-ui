## Usage

```tsx
import { useFilter, useListCollection } from "@ark-ui/react";
import {
  ModelSelector,
  ModelSelectorContent,
  ModelSelectorEmpty,
  ModelSelectorGroup,
  ModelSelectorInput,
  ModelSelectorItem,
  ModelSelectorLabel,
  ModelSelectorList,
  ModelSelectorTrigger,
} from "@/components/ui/model-selector";
```

```tsx
const { contains } = useFilter({ sensitivity: "base" });
const { collection, filter } = useListCollection({
  filter: contains,
  groupBy: (item) => item.group,
  initialItems: [
    { group: "OpenAI", label: "GPT-4.1", value: "gpt-4.1" },
    { group: "Anthropic", label: "Claude Sonnet 4", value: "claude-sonnet-4" },
  ],
});

<ModelSelector
  collection={collection}
  onInputValueChange={({ inputValue }) => filter(inputValue)}
>
  <ModelSelectorTrigger />
  <ModelSelectorContent>
    <ModelSelectorInput placeholder="Search models" />
    <ModelSelectorList>
      <ModelSelectorEmpty />
      {collection.group().map(([group, items]) => (
        <ModelSelectorGroup key={group}>
          <ModelSelectorLabel>{group}</ModelSelectorLabel>
          {items.map((item) => (
            <ModelSelectorItem item={item} key={item.value}>
              {item.label}
            </ModelSelectorItem>
          ))}
        </ModelSelectorGroup>
      ))}
    </ModelSelectorList>
  </ModelSelectorContent>
</ModelSelector>
```

## API Reference

### ModelSelector

Root [Combobox](/docs/components/combobox). Holds the selected model value.

| Prop | Type | Default |
| --- | --- | --- |
| `collection` | `ListCollection<T>` | **required** |
| `value` | `string \| string[]` | `-` |
| `defaultValue` | `string \| string[]` | `-` |
| `onValueChange` | `(details: ValueChangeDetails<T>) => void` | `-` |
| `onInputValueChange` | `(details: InputValueChangeDetails) => void` | `-` |
| `open` | `boolean` | `-` |
| `onOpenChange` | `(details: OpenChangeDetails) => void` | `-` |
| `disabled` | `boolean` | `-` |
| `selectionBehavior` | `"replace" \| "clear" \| "preserve"` | `"clear"` |

### ModelSelectorTrigger

Opens the picker. Shows the selected item via Combobox `valueAsString`. Pass children to override. Pass `showTrigger` to show the chevron. Focus moves to the search field when the panel opens.

| Prop | Type | Default |
| --- | --- | --- |
| `variant` | `"outline" \| "ghost"` | `"outline"` |
| `size` | `"xs" \| "sm" \| "md" \| "lg" \| "xl" \| "icon-xs" \| "icon-sm" \| "icon-md" \| "icon-lg" \| "icon-xl"` | `"sm"` |
| `showTrigger` | `boolean` | `false` |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### ModelSelectorContent

Holds the search field and list.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `asChild` | `boolean` | `false` |

### ModelSelectorInput

Combobox search field. Filter the collection from `onInputValueChange` on the root.

| Prop | Type | Default |
| --- | --- | --- |
| `placeholder` | `string` | `-` |
| `className` | `string` | `-` |

### ModelSelectorList

Scrollable list of model options.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |

### ModelSelectorEmpty

Shown when the list has no matching models.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
| `children` | `ReactNode` | `No models found.` |

### ModelSelectorGroup

Groups related models under a label.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |

### ModelSelectorLabel

Label for a model group.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |

### ModelSelectorItem

Selectable model option. Pass the collection item.

| Prop | Type | Default |
| --- | --- | --- |
| `item` | `T` | **required** |
| `className` | `string` | `-` |

### ModelSelectorName

Primary label for a model. Truncates when space is tight.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |

### ModelSelectorDescription

Supporting meta under the name, such as context window.

| Prop | Type | Default |
| --- | --- | --- |
| `className` | `string` | `-` |
